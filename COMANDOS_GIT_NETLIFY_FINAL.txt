COLE ESTE BLOCO NO TERMINAL SOMENTE DEPOIS DE ENTRAR NA PASTA EXTRAÍDA PAIDEGUA_FINAL_CATEGORIZADO:

deploy_paidegua_categorizado() {
  SRC="$PWD"
  REPO="$HOME/Downloads/PAIDEGUA_DEPLOY_CATEGORIZADO"
  GITHUB="https://github.com/Gerrard013/Cardapio_premium_Pai_D-Egua.git"

  echo "=============================================="
  echo "PAI D'ÉGUA — DEPLOY FINAL CATEGORIZADO"
  echo "Origem: $SRC"
  echo "=============================================="

  [ -f "$SRC/index.html" ] || { echo "ERRO: index.html não está nesta pasta."; return 1; }
  [ -f "$SRC/netlify.toml" ] || { echo "ERRO: netlify.toml não está nesta pasta."; return 1; }
  [ -d "$SRC/assets/images" ] || { echo "ERRO: assets/images não existe."; return 1; }
  [ -f "$SRC/scripts/verify-project.mjs" ] || { echo "ERRO: verificador ausente."; return 1; }

  cd "$SRC" || return 1
  node --check js/app.js || return 1
  node --check js/menu.js || return 1
  node scripts/verify-project.mjs || return 1

  rm -rf "$REPO"
  git clone "$GITHUB" "$REPO" || return 1
  cd "$REPO" || return 1
  git checkout main || return 1
  git pull --ff-only origin main || return 1

  COMMIT_ANTERIOR="$(git rev-parse HEAD)"
  echo "$COMMIT_ANTERIOR" > "$HOME/Downloads/PAIDEGUA_COMMIT_ANTES_CATEGORIAS.txt"

  rsync -a --delete \
    --exclude='.git' \
    --exclude='__MACOSX' \
    --exclude='.DS_Store' \
    "$SRC/" "$REPO/"

  [ -f index.html ] || { echo "ERRO: index.html não ficou na raiz."; return 1; }
  grep -Eq 'publish[[:space:]]*=[[:space:]]*"\."' netlify.toml || { echo "ERRO: Netlify não está publicando a raiz."; return 1; }

  node --check js/app.js || return 1
  node --check js/menu.js || return 1
  node scripts/verify-project.mjs || return 1

  echo
  echo "ALTERAÇÕES QUE SERÃO PUBLICADAS:"
  git status --short
  git diff --stat || true

  git add -A
  if git diff --cached --quiet; then
    git commit --allow-empty -m "chore: força deploy final categorizado Pai D'Égua" || return 1
  else
    git commit -m "feat: cardápio final categorizado Pai D'Égua" || return 1
  fi

  git push origin main || return 1

  echo "=============================================="
  echo "GITHUB ATUALIZADO — NETLIFY EM DEPLOY"
  echo "Commit anterior: $COMMIT_ANTERIOR"
  echo "=============================================="
  git log -1 --oneline
  open "https://app.netlify.com/projects/paidegua-cardapio-oficial/deploys"
  cd "$SRC" || true
}

deploy_paidegua_categorizado
unset -f deploy_paidegua_categorizado
