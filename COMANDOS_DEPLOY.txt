1) Baixe o ZIP final para a pasta Downloads.

2) Cole no Terminal do VS Code:

ZIP="$HOME/Downloads/PAIDEGUA_CARDAPIO_PREMIUM_ENTREGA_DEFINITIVA.zip"
EXTRAIDO="$HOME/Downloads/PAIDEGUA_ENTREGA_DEFINITIVA"
REPO="$HOME/Downloads/PAIDEGUA_GIT_DEPLOY"

rm -rf "$EXTRAIDO"
mkdir -p "$EXTRAIDO"
unzip -o "$ZIP" -d "$EXTRAIDO"

NOVO="$EXTRAIDO/PAIDEGUA_PREMIUM_FINAL"

cd "$REPO"
git checkout main
git pull --ff-only origin main

rsync -av --delete --exclude='.git' "$NOVO/" "$REPO/"

git add -A
git commit -m "feat: entrega definitiva Cardápio Premium Pai D'Égua"
git push origin main

3) Acompanhe:
open "https://app.netlify.com/projects/paidegua-cardapio-oficial/deploys"

4) Abra o site:
open "https://paidegua-cardapio-oficial.netlify.app/?v=entrega-definitiva"
